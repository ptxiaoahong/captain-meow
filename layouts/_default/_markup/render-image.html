{{- $image := .Page.Resources.GetMatch (printf "%s" (.Destination | safeURL)) -}} {{- $Permalink := .Destination | relURL | safeURL -}} {{- $alt := .PlainText | safeHTML -}} {{- $Width := 0 -}} {{- $Height := 0 -}} {{- $Srcset := "" -}} {{- $galleryImage := false -}} {{- $assetsImage := false -}} {{- $staticImage := false -}} {{- $notSVG := ne (path.Ext .Destination) ".svg" -}} {{- if $image -}} {{- /* 页面资源中的图片 */ -}} {{- $Permalink = $image.RelPermalink -}} {{- if $notSVG -}} {{- $Width = $image.Width -}} {{- $Height = $image.Height -}} {{- $galleryImage = true -}} {{- /* 使用原始图片，不压缩 */ -}} {{- end -}} {{- else -}} {{- /* 无论是否找到资源，只要不是 SVG，就标记为 galleryImage */ -}} {{- if $notSVG -}} {{- $galleryImage = true -}} {{- end -}} {{- /* 尝试从 assets/images 获取图片尺寸 */ -}} {{- $assetsPath := .Destination | strings.TrimPrefix "/" -}} {{- $assetsImage = resources.Get (printf "images/%s" $assetsPath) -}} {{- if $assetsImage -}} {{- $Permalink = $assetsImage.RelPermalink -}} {{- $Width = $assetsImage.Width -}} {{- $Height = $assetsImage.Height -}} {{- else -}} {{- /* 尝试从 static 获取图片尺寸 */ -}} {{- $staticPath := .Destination | strings.TrimPrefix "/" -}} {{- $staticImage = resources.Get $staticPath -}} {{- if $staticImage -}} {{- $Permalink = $staticImage.RelPermalink -}} {{- $Width = $staticImage.Width -}} {{- $Height = $staticImage.Height -}} {{- end -}} {{- end -}} {{- end -}}
