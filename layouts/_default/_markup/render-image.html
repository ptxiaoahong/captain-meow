{{- $image := .Page.Resources.GetMatch (printf "%s" (.Destination | safeURL)) -}} {{- $Permalink := .Destination | relURL | safeURL -}} {{- $alt := .PlainText | safeHTML -}} {{- $Width := 0 -}} {{- $Height := 0 -}} {{- $Srcset := "" -}} {{- $galleryImage := false -}} {{- $assetsImage := false -}} {{- $staticImage := false -}} {{- $notSVG := ne (path.Ext .Destination) ".svg" -}} {{- if $image -}} {{- /* 页面资源中的图片 */ -}} {{- $Permalink = $image.RelPermalink -}} {{- if $notSVG -}} {{- $Width = $image.Width -}} {{- $Height = $image.Height -}} {{- $galleryImage = true -}} {{- /* 使用原始图片，不压缩 */ -}} {{- end -}} {{- else -}} {{- /* 优先从 assets/images 获取图片（支持自动压缩） */ -}} {{- $assetsPath := .Destination | strings.TrimPrefix "/" -}} {{- $assetsImage = resources.Get (printf "images/%s" $assetsPath) -}} {{- if $assetsImage -}} {{- /* 从 assets/images 找到图片 */ -}} {{- $Permalink = $assetsImage.RelPermalink -}} {{- if $notSVG -}} {{- $Width = $assetsImage.Width -}} {{- $Height = $assetsImage.Height -}} {{- $galleryImage = true -}} {{- /* 使用原始图片，不压缩 */ -}} {{- end -}} {{- else -}} {{- /* 后备：从 static 目录获取 */ -}} {{- $staticPath := .Destination | strings.TrimPrefix "/" -}} {{- $staticImage = resources.Get $staticPath -}} {{- if $staticImage -}} {{- $Permalink = $staticImage.RelPermalink -}} {{- if $notSVG -}} {{- $Width = $staticImage.Width -}} {{- $Height = $staticImage.Height -}} {{- $galleryImage = true -}} {{- /* 使用原始图片，不压缩 */ -}} {{- end -}} {{- else -}} {{- /* 如果都找不到，使用直接路径，仍然标记为画廊图片 */ -}} {{- if $notSVG -}} {{- $galleryImage = true -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
