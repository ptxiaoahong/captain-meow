{{/* 用法：{{ partial "icon" (dict "name" "calendar" "class" "icon") }} name 可选：calendar / clock / mail / wechat / arrow-left / arrow-up / chat / close */}} {{- $name := .name -}} {{- $class := default "icon" .class -}} {{- if eq $name "calendar" -}} {{- else if eq $name "clock" -}} {{- else if eq $name "mail" -}} {{- else if eq $name "wechat" -}} {{- else if eq $name "arrow-left" -}} {{- else if eq $name "arrow-up" -}} {{- else if eq $name "chat" -}} {{- else if eq $name "menu" -}} {{- else if eq $name "close" -}} {{- end -}}
